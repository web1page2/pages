const SPREADSHEET_NAME = "data"; const HEADERS = [ "roll", "name", "photo", "email", "phone number", "whatsapp number", "district", "address", "college name", "facebook link", "codeforces", "timeStamp", "ip" ]; // Upload base64 image to Drive and return a raw download link for the IMAGE() formula function uploadPhotoToDrive(photoFile, roll) { try { const folderName = "Student_Photos"; let folder; const folders = DriveApp.getFoldersByName(folderName); if (folders.hasNext()) { folder = folders.next(); } else { folder = DriveApp.createFolder(folderName); } const decoded = Utilities.base64Decode(photoFile.base64Data); const blob = Utilities.newBlob(decoded, photoFile.mimeType, `photo_${roll}_${Date.now()}`); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); // CRUCIAL FIX: 'export=download' allows Google Sheets to fetch the raw image bytes return `https://drive.google.com/uc?export=download&id=${file.getId()}`; } catch (err) { Logger.log("Drive Upload Error: " + err.toString()); return ""; } } // 1. Get or create sheet function getOrCreateSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SPREADSHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SPREADSHEET_NAME); sheet.appendRow(HEADERS); sheet.getRange(1, 1, 1, HEADERS.length).setFontWeight("bold"); } return sheet; } // 2. HTTP POST Handler function doPost(e) { try { if (!e || !e.postData || !e.postData.contents) { throw new Error("Empty request payload received."); } const requestData = JSON.parse(e.postData.contents); const action = requestData.action; const payload = requestData.data || {}; const clientIp = requestData.clientIp || "0.0.0.0"; let result; if (action === "CREATE" || action === "UPDATE") { result = createOrUpdateData(payload, clientIp); } else if (action === "READ") { result = readData(payload.roll); } else if (action === "DELETE") { result = deleteData(payload.roll); } else { throw new Error("Invalid action specified."); } return ContentService.createTextOutput(JSON.stringify({ status: "success", result: result })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } // 3. CREATE or UPDATE Logic function createOrUpdateData(data, clientIp) { if (!data.roll) throw new Error("'roll' field is mandatory."); const sheet = getOrCreateSheet(); const rows = sheet.getDataRange().getValues(); const formulas = sheet.getDataRange().getFormulas(); // Need formulas to preserve existing images const rollIndex = 0; let targetRowIndex = -1; let currentRowData = []; let currentRowFormulas = []; let existingIpList = []; for (let i = 1; i < rows.length; i++) { if (String(rows[i][rollIndex]) === String(data.roll)) { targetRowIndex = i + 1; currentRowData = rows[i]; currentRowFormulas = formulas[i]; let rawIp = currentRowData[12]; if (rawIp) { try { let parsed = JSON.parse(rawIp); existingIpList = Array.isArray(parsed) ? parsed : [String(rawIp)]; } catch (e) { existingIpList = [String(rawIp)]; } } break; } } if (clientIp && clientIp !== "0.0.0.0" && !existingIpList.includes(clientIp)) { existingIpList.push(clientIp); } const timestamp = new Date().toISOString(); let photoUrl = ""; if (data.photoFile && data.photoFile.base64Data) { photoUrl = uploadPhotoToDrive(data.photoFile, data.roll); } let activeRow; if (targetRowIndex > -1) { // If no new photo, keep the existing formula/URL const existingPhotoValue = currentRowFormulas[2] || currentRowData[2] || ""; const finalPhotoField = photoUrl ? `=IMAGE("${photoUrl}")` : existingPhotoValue; const updatedRow = [ data.roll, data.name || currentRowData[1], finalPhotoField, data.email || currentRowData[3], data.phone || currentRowData[4], data.whatsapp || currentRowData[5], data.district || currentRowData[6], data.address || currentRowData[7], data.college || currentRowData[8], data.facebook || currentRowData[9], data.codeforces || currentRowData[10], timestamp, JSON.stringify(existingIpList) ]; sheet.getRange(targetRowIndex, 1, 1, updatedRow.length).setValues([updatedRow]); activeRow = targetRowIndex; } else { let newIpList = (clientIp && clientIp !== "0.0.0.0") ? [clientIp] : []; const finalPhotoField = photoUrl ? `=IMAGE("${photoUrl}")` : ""; const newRow = [ data.roll, data.name || "", finalPhotoField, data.email || "", data.phone || "", data.whatsapp || "", data.district || "", data.address || "", data.college || "", data.facebook || "", data.codeforces || "", timestamp, JSON.stringify(newIpList) ]; sheet.appendRow(newRow); activeRow = sheet.getLastRow(); } // Format the row/column so the cell image is actually big enough to see sheet.setRowHeight(activeRow, 80); if (sheet.getColumnWidth(3) < 80) { sheet.setColumnWidth(3, 80); } sheet.getRange(activeRow, 3).setHorizontalAlignment("center").setVerticalAlignment("middle"); return { message: targetRowIndex > -1 ? "Record updated successfully" : "Record created successfully", roll: data.roll }; } // 4. READ Logic function readData(roll) { const sheet = getOrCreateSheet(); const rows = sheet.getDataRange().getValues(); const formulas = sheet.getDataRange().getFormulas(); if (roll) { for (let i = 1; i < rows.length; i++) { if (String(rows[i][0]) === String(roll)) { return parseRowToObj(rows[i], formulas[i]); } } throw new Error("Record not found for roll: " + roll); } else { return rows.slice(1).map((row, idx) => parseRowToObj(row, formulas[idx + 1])); } } // 5. DELETE Logic function deleteData(roll) { if (!roll) throw new Error("'roll' is required for deletion."); const sheet = getOrCreateSheet(); const rows = sheet.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (String(rows[i][0]) === String(roll)) { sheet.deleteRow(i + 1); return { message: "Record deleted successfully", roll: roll }; } } throw new Error("Record not found for roll: " + roll); } // Helper: Extract formula URL and convert row array to JSON response function parseRowToObj(row, formulaRow) { let photoValue = row[2]; let photoFormula = formulaRow[2]; let finalPhotoUrl = photoValue; // Extract pure URL from =IMAGE("url") so frontend can display it if (photoFormula && photoFormula.toUpperCase().startsWith("=IMAGE")) { const match = photoFormula.match(/=IMAGE\("(.*?)"/i); if (match && match[1]) { finalPhotoUrl = match[1]; } } let parsedIp = []; try { let rawIp = row[12]; if (rawIp && typeof rawIp === 'string' && rawIp.startsWith('[')) { parsedIp = JSON.parse(rawIp); } else if (rawIp) { parsedIp = [String(rawIp)]; } } catch(e) { parsedIp = [String(row[12])]; } return { roll: row[0], name: row[1], photo: finalPhotoUrl, email: row[3], phone: row[4], whatsapp: row[5], district: row[6], address: row[7], college: row[8], facebook: row[9], codeforces: row[10], timeStamp: row[11], ip: parsedIp }; }
